
var gravity:Number = 1;
var wind:Number = 0;
var airFriction:Number = .97;
var ground:Number = 700;

var particles:Object = new Object();

var particleNum:Number = 0;

var maxParticles:Number = 18;

var maxDistance:Number = 100;

var drawLayer:Sprite = new Sprite();
addChild(drawLayer);

function spawnParticle (event:TimerEvent) {
	
	var particle:Sprite = new Sprite();
	
	particle.graphics.beginFill(0xFFFFFF, 1);
	particle.graphics.drawCircle(0, 0, 4);
	particle.graphics.beginFill(0x222222, 1);
	particle.graphics.drawCircle(0, 0, 2);
	particle.graphics.endFill();
	
	particle.x = mouseX;
	particle.y = mouseY;
	
	addChild(particle);
	
	var mag:Number = Math.random()*2;
	var dir:Number = Math.random()*Math.PI*2;
	
	particles["p"+particleNum] = {x:mouseX, y:mouseY, ref:particle,
									xSpeed:Math.cos(dir)*mag, 
									ySpeed:Math.sin(dir)*mag}
									
		
	if (particleNum>=maxParticles) {
		removeChild(particles["p"+(particleNum-maxParticles)].ref);
		delete particles["p"+(particleNum-maxParticles)];
	}
	
	particleNum++;
		
		
}



function moveParticles (event:TimerEvent){
	
	//if (Math.random()<.05) wind = Math.random()*2-1;
	
	drawLayer.graphics.clear();
	drawLayer.graphics.lineStyle(0, 0xFFFFFF, 1);
	
	var min:Number = particleNum-maxParticles;
	if (min<0) min = 0;
	
	//trace("particleNum: "+particleNum);
	//trace("min: "+min);
	
	for (var c=particleNum-1; c>min; c--) {
		
		var p:Object  = particles["p"+c];
		var nb:Object = particles["p"+(c-1)];
		
		p.xSpeed += wind;
		p.ySpeed += gravity;
		
		p.xSpeed *= airFriction;
		p.ySpeed *= airFriction;
		
		p.x += p.xSpeed;
		p.y += p.ySpeed;
		
		if (p.y > ground) {
			p.y = ground;
			p.ySpeed *= -1;
		}
		
		var X:Number = p.x - nb.x;
		var Y:Number = p.y - nb.y;
		
		var Distance:Number = Math.sqrt(X*X+Y*Y);
		var Angle:Number = Math.atan2(Y, X);
		
		if (Distance>maxDistance) {
			p.x = nb.x + maxDistance * Math.cos(Angle);
			p.y = nb.y + maxDistance * Math.sin(Angle);
		}
		
		p.ref.x = p.x;
		p.ref.y = p.y;
		
		if (c == particleNum-1) {
			drawLayer.graphics.moveTo(p.x, p.y);
		} else {
			drawLayer.graphics.lineTo(p.x, p.y);
		}
	}
	/*for (var b=min; b<particleNum-1; b++) {

		if (b == min) {
			drawLayer.graphics.moveTo(p.x, p.y);
		} else {
			drawLayer.graphics.lineTo(p.x, p.y);
		}
		
	}*/

		
}

var myTimer:Timer = new Timer(1000/10, 0);
myTimer.addEventListener("timer", spawnParticle);
myTimer.start();

var myTimerB:Timer = new Timer(1000/30, 0);
myTimerB.addEventListener("timer", moveParticles);
myTimerB.start();
